Replace the initWindowSection() function. Adjust the timing so the crossfade happens perfectly between 35% and 60% of the scroll, ensuring both elements are locked at top: 0 while they fade.

JavaScript
(function initWindowSection() {
  const section = document.getElementById('window-section');
  const cloudsImg = document.getElementById('clouds-img');
  const windowImg = document.getElementById('window-img');
  const destinations = document.getElementById('destinations-section');
  if (!section || !cloudsImg || !windowImg) return;

  gsap.to(cloudsImg, {
    y: '-10%',
    x: '-10%',
    ease: 'none',
    scrollTrigger: {
      trigger: section,
      start: 'top top',
      end: '60% top',
      scrub: 1,
    }
  });

  gsap.to(windowImg, {
    scale: 25,
    ease: 'power2.in',
    scrollTrigger: {
      trigger: section,
      start: 'top top',
      end: '50% top',
      scrub: 1.2,
    }
  });

  // Fades out exactly while locked in place
  gsap.to([cloudsImg, windowImg], {
    opacity: 0,
    ease: 'power2.inOut',
    scrollTrigger: {
      trigger: section,
      start: '35% top', // Starts fading after window zooms out a bit
      end: '60% top',
      scrub: 1,
    }
  });

  // Fades in seamlessly right underneath
  gsap.fromTo('#destinations-section',
    { opacity: 0 },
    {
      opacity: 1,
      ease: 'power2.inOut',
      scrollTrigger: {
        trigger: section,
        start: '35% top',
        end: '60% top', // Perfectly synchronized
        scrub: 1,
      }
    }
  );
})();
