(function initWindowSection() {
  const section = document.getElementById('window-section');
  const cloudsImg = document.getElementById('clouds-img');
  const windowImg = document.getElementById('window-img');
  const cineText = document.getElementById('cinematic-text');
  if (!section || !cloudsImg || !windowImg) return;

  // Create a master timeline for the window section
  const tl = gsap.timeline({
    scrollTrigger: {
      trigger: section,
      start: 'top top',
      end: 'bottom bottom',
      scrub: 1,
    }
  });

  // 1. Move clouds slightly
  tl.to(cloudsImg, { y: '-10%', x: '-10%', ease: 'none', duration: 2 }, 0);

  // 2. Zoom window 
  tl.to(windowImg, { scale: 25, ease: 'power2.in', duration: 2 }, 0);

  // 3. Fade out clouds and window early
  tl.to([cloudsImg, windowImg], { opacity: 0, ease: 'power2.inOut', duration: 1 }, 1.5);

  // 4. Fade IN the cinematic text in the black void
  tl.to(cineText, { opacity: 1, y: -20, ease: 'power1.out', duration: 1 }, 2.5);

  // 5. Fade OUT the cinematic text before leaving the section
  tl.to(cineText, { opacity: 0, y: -40, ease: 'power1.in', duration: 1 }, 4.0);
})();
