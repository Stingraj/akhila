JavaScript — add this inside the second <script> block, after initExploration():
js(function initWindowSection() {
  const section = document.getElementById('window-section');
  const cloudsImg = document.getElementById('clouds-img');
  const windowImg = document.getElementById('window-img');
  if (!section || !cloudsImg || !windowImg) return;

  // Clouds drift right to left continuously
  gsap.to(cloudsImg, {
    x: '-50%',
    ease: 'none',
    scrollTrigger: {
      trigger: section,
      start: 'top top',
      end: 'bottom bottom',
      scrub: 1,
    }
  });

  // Window zooms in as user scrolls — scale 15 fills screen for 16:9 ratio
  gsap.to(windowImg, {
    scale: 15,
    ease: 'power2.in',
    scrollTrigger: {
      trigger: section,
      start: 'top top',
      end: '90% bottom',
      scrub: 1.5,
    }
  });

  // Background color transitions from sky blue to dark as window fills screen
  gsap.to('#window-sticky', {
    backgroundColor: '#0a0806',
    ease: 'power1.in',
    scrollTrigger: {
      trigger: section,
      start: '60% top',
      end: '90% bottom',
      scrub: 1,
    }
  });
})();
