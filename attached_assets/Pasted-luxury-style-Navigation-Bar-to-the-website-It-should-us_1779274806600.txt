3. JS Update (Smart Scroll Logic):
Add this simple logic at the end of the script tag to handle the hide/show functionality using the existing Lenis instance. Put this inside the existing setup:

JavaScript
// Add this near the end of your script, after initializing Lenis
(function initNavbar() {
  const nav = document.getElementById('main-nav');
  if (!nav) return;
  
  let lastScrollY = 0;
  
  lenis.on('scroll', (e) => {
    const currentScrollY = e.animatedScroll;
    
    if (currentScrollY > lastScrollY && currentScrollY > 100) {
      // Scrolling down -> hide navbar
      nav.classList.add('hidden');
    } else {
      // Scrolling up -> show navbar
      nav.classList.remove('hidden');
    }
    lastScrollY = currentScrollY;
  });
})();
