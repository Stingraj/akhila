  if (!canvas || typeof createGlobe === 'undefined') return;

  const globe = createGlobe(canvas, {
    devicePixelRatio: Math.min(window.devicePixelRatio, 2),
    width: canvas.offsetWidth * 2,
    height: canvas.offsetHeight * 2,
    phi: 0,
    theta: 0.3,
    dark: 0,
    diffuse: 1.2,
    mapSamples: 16000,
    mapBrightness: 6,
    baseColor: [1, 1, 1],
    markerColor: [0.1, 0.3, 0.7],
    glowColor: [1, 1, 1],
    markers: [
      { location: [51.5, -116.1], size: 0.08 },
      { location: [35.6, 139.7], size: 0.08 },
      { location: [40.6, 14.6], size: 0.08 },
      { location: [3.2, 73.2], size: 0.08 },
    ],
    onRender: (state) => {
      state.phi = state.phi + 0.003;
      state.width = canvas.offsetWidth * 2;
      state.height = canvas.offsetHeight * 2;
    }
  });
})();
