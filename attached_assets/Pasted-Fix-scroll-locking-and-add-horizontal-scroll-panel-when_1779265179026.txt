1. Fix scroll lock — replace the openDetail and closeDetail functions entirely:
jsfunction openDetail(panelIndex) {
  const panel = panels[panelIndex];
  const roomOverlay = panel.querySelector('.room-overlay');
  const leftImg = panel.querySelector('.panel-left > img');
  openStates[panelIndex] = true;

  const tl = gsap.timeline();
  tl.to(roomOverlay, { opacity: 1, duration: 0.3, ease: 'power2.out' })
    .call(() => { panel.classList.add('detail-open'); }, null, 0.4)
    .to(leftImg, { scale: 0.55, duration: 0.2, ease: 'power2.inOut' }, 0.4)
    .to(roomOverlay, { opacity: 0, duration: 0.3, ease: 'power2.in' }, 0.7)
    .call(() => {
      // Hard lock all scroll
      const scrollY = window.scrollY;
      document.body.style.position = 'fixed';
      document.body.style.top = `-${scrollY}px`;
      document.body.style.width = '100%';
      document.body.dataset.scrollY = scrollY;
      if (lenis) lenis.stop();
    }, null, 1.0);
}

function closeDetail(panelIndex) {
  const panel = panels[panelIndex];
  const roomOverlay = panel.querySelector('.room-overlay');
  const leftImg = panel.querySelector('.panel-left > img');
  openStates[panelIndex] = false;

  const tl = gsap.timeline();
  tl.to(roomOverlay, { opacity: 1, duration: 0.3, ease: 'power2.out' })
    .to(leftImg, { scale: 1, duration: 0.2, ease: 'power2.inOut' }, 0.3)
    .call(() => { panel.classList.remove('detail-open'); }, null, 0.4)
    .to(roomOverlay, { opacity: 0, duration: 0.3, ease: 'power2.in' }, 0.5)
    .call(() => {
      // Restore scroll position exactly
      const scrollY = parseInt(document.body.dataset.scrollY || '0');
      document.body.style.position = '';
      document.body.style.top = '';
      document.body.style.width = '';
      window.scrollTo(0, scrollY);
      if (lenis) lenis.start();
    }, null, 0.8);
}
