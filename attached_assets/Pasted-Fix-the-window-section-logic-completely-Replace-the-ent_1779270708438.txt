js(function initWindowSection() {
  const section = document.getElementById('window-section');
  const cloudsImg = document.getElementById('clouds-img');
  const windowImg = document.getElementById('window-img');
  const destinations = document.getElementById('destinations-section');
  if (!section || !cloudsImg || !windowImg) return;

  // Clouds move 20% left and 20% up
  gsap.to(cloudsImg, {
    y: '-20%',
    x: '-20%',
    ease: 'none',
    scrollTrigger: {
      trigger: section,
      start: 'top top',
      end: 'bottom bottom',
      scrub: 1,
    }
  });

  // Window zooms immediately from start
  gsap.to(windowImg, {
    scale: 10,
    ease: 'power2.in',
    scrollTrigger: {
      trigger: section,
      start: 'top top',
      end: '60% bottom',
      scrub: 1.2,
    }
  });

  // Clouds fade out as zoom completes
  gsap.to(cloudsImg, {
    opacity: 0,
    ease: 'power1.in',
    scrollTrigger: {
      trigger: section,
      start: '40% top',
      end: '65% bottom',
      scrub: 1,
    }
  });

  // Window also fades out
  gsap.to(windowImg, {
    opacity: 0,
    ease: 'power1.in',
    scrollTrigger: {
      trigger: section,
      start: '50% top',
      end: '65% bottom',
      scrub: 1,
    }
  });

  // Background stays dark #0a0806 — no color animation needed

  // Destinations fade in over the dark background
  gsap.fromTo('#destinations-section',
    { opacity: 0 },
    {
      opacity: 1,
      ease: 'power1.out',
      scrollTrigger: {
        trigger: section,
        start: '60% top',
        end: '80% bottom',
        scrub: 1,
      }
    }
  );
})();
