Replace the current click handler with this GSAP timeline sequence:
jsfunction openDetail(panelIndex) {
  const panel = panels[panelIndex];
  const overlay = panel.querySelector('.room-overlay');
  const leftImg = panel.querySelector('.panel-left img');

  const tl = gsap.timeline();

  // 0.0 - 0.3s: overlay fades in over right panel
  tl.to(overlay, {
    opacity: 1,
    duration: 0.3,
    ease: 'power2.out',
    pointerEvents: 'none'
  })

  // 0.4s: left image starts shrinking + bg color + text appear
  .call(() => {
    // trigger left panel detail state
    panel.classList.add('detail-open');
  }, null, 0.4)

  .to(leftImg, {
    scale: 0.55,
    duration: 0.2,
    ease: 'power2.inOut'
  }, 0.4)

  // 0.7s: overlay fades out revealing room image
  .to(overlay, {
    opacity: 0,
    duration: 0.3,
    ease: 'power2.in'
  }, 0.7)

  // lock scroll after animation
  .call(() => {
    document.body.style.overflow = 'hidden';
    lenis.stop();
  }, null, 1.0);
}
Close animation (reverse):
jsfunction closeDetail(panelIndex) {
  const panel = panels[panelIndex];
  const overlay = panel.querySelector('.room-overlay');
  const leftImg = panel.querySelector('.panel-left img');

  const tl = gsap.timeline();

  tl.to(overlay, { opacity: 1, duration: 0.3, ease: 'power2.out' })
  .to(leftImg, { scale: 1, duration: 0.2, ease: 'power2.inOut' }, 0.3)
  .call(() => { panel.classList.remove('detail-open'); }, null, 0.4)
  .to(overlay, { opacity: 0, duration: 0.3, ease: 'power2.in' }, 0.5)
  .call(() => {
    document.body.style.overflow = '';
    lenis.start();
  }, null, 0.8);
}
